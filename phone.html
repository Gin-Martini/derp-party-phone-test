// server.js — tiny relay w/ proper host<->player routing + logs const express = require('express'); const cors = require('cors'); const { WebSocketServer } = require('ws'); const { customAlphabet } = require('nanoid'); const app = express(); app.use(cors()); app.use(express.json()); const nanoRoom = customAlphabet('ABCDEFGHJKLMNPQRSTUVWXYZ23456789', 6); const nanoId = customAlphabet('abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789', 8); // In-memory rooms const rooms = new Map(); // roomId => { host: WebSocket|null, players: Map } function ensureRoom(roomId) { let r = rooms.get(roomId); if (!r) { r = { host: null, players: new Map() }; rooms.set(roomId, r); } return r; } // --- HTTP: health + room management --- app.get('/healthz', (_req, res) => res.status(200).send('ok')); app.post('/rooms', (_req, res) => { const roomId = nanoRoom(); ensureRoom(roomId); console.log('[HTTP] create room', roomId); res.json({ roomId }); }); app.post('/rooms/:roomId/join', (req, res) => { const roomId = (req.params.roomId || '').toUpperCase(); if (!rooms.has(roomId)) return res.status(404).json({ error: 'no_room' }); const playerId = nanoId(); console.log('[HTTP] join', roomId, '->', playerId); res.json({ playerId }); }); const server = app.listen(process.env.PORT || 8787, () => { console.log(`Relay listening on http://0.0.0.0:${process.env.PORT || 8787}`); }); // --- WebSocket --- const wss = new WebSocketServer({ server, path: '/socket' }); function send(ws, obj) { try { ws.readyState === 1 && ws.send(JSON.stringify(obj)); } catch {} } wss.on('connection', (ws) => { ws.meta = { role: 'UNK', roomId: null, playerId: null, name: null }; ws.on('message', (data) => { let msg; try { msg = JSON.parse(data.toString()); } catch { return; } const now = Date.now(); // Identify connections if (msg.type === 'HELLO_HOST') { const roomId = (msg.roomId || '').toUpperCase(); const room = ensureRoom(roomId); if (room.host && room.host !== ws) { try { room.host.close(); } catch {} } room.host = ws; ws.meta = { role: 'HOST', roomId }; console.log('[WS] HOST hello', roomId); send(ws, { type: 'ROOM_OPEN', roomId, ts: now }); return; } if (msg.type === 'HELLO_PLAYER') { const roomId = (msg.roomId || '').toUpperCase(); const playerId = msg.playerId || nanoId(); const name = msg.name || 'Player'; const room = ensureRoom(roomId); ws.meta = { role: 'PLAYER', roomId, playerId, name }; room.players.set(playerId, ws); console.log('[WS] PLAYER hello', roomId, playerId, name); send(ws, { type: 'WELCOME', roomId, playerId, name }); if (room.host) send(room.host, { type: 'PLAYER_JOINED', roomId, playerId, name, ts: now }); return; } // Heartbeat if (msg.type === 'PING') { send(ws, { type: 'PONG', ts: now }); return; } // Host -> all players (tunnel state) if (ws.meta.role === 'HOST' && msg.type === 'BROADCAST_STATE') { const room = rooms.get(ws.meta.roomId); if (!room) return; // console.log('[HOST->PLAYERS]', ws.meta.roomId, 'state.type=', msg?.state?.type); room.players.forEach(pws => send(pws, { type: 'STATE', state: msg.state, ts: now })); return; } // Player -> Host (forward intents) if (ws.meta.role === 'PLAYER') { const room = rooms.get(ws.meta.roomId); if (!room || !room.host) return; if (msg.type === 'CARD_INTENT' || msg.type === 'ANSWER_SUBMIT' || msg.type === 'READY_TOGGLE') { const fwd = { ...msg, roomId: ws.meta.roomId, senderId: ws.meta.playerId, playerId: ws.meta.playerId, recvTs: now, role: 'PLAYER' }; console.log('[PLAYER->HOST]', fwd.type, fwd.playerId, 'room', fwd.roomId); send(room.host, fwd); return; } return; } }); ws.on('close', () => { const { role, roomId, playerId } = ws.meta || {}; if (!roomId) return; const room = rooms.get(roomId); if (!room) return; if (role === 'HOST' && room.host === ws) { room.host = null; console.log('[WS] HOST left', roomId); room.players.forEach(pws => send(pws, { type: 'STATE', state: { type: 'TEXT', message: 'Host disconnected' }, ts: Date.now() })); } else if (role === 'PLAYER') { if (room.players.get(playerId) === ws) room.players.delete(playerId); console.log('[WS] PLAYER left', roomId, playerId); if (room.host) send(room.host, { type: 'PLAYER_LEFT', roomId, playerId, ts: Date.now() }); } }); });
